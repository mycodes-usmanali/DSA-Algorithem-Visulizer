Linear Search Algorithm

Time Complexity: O(n) 
function linearSearch(arr, target) {
    for (let i = 0; i < arr.length; i++) {
        if (arr[i] === target) {
            return i; // Found at index i
        }
    }
    
    return -1; // Not found in array
}
